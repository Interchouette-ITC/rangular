1//! Proc macros for rangular component templates.
2//!
3//! Expands to `include!` of AOT output under `OUT_DIR/rangular/`. The consuming
4//! crate's `build.rs` must call `rangular_aot::compile_named` and write
5//! `{fn_name}.rs` there before compiling the crate that invokes this macro.
6
7#![forbid(unsafe_code)]
8
9use std::path::PathBuf;
10
11use proc_macro::TokenStream;
12use quote::quote;
13use syn::{parse_macro_input, LitStr};
14
15#[proc_macro]
16pub fn rangular_template(input: TokenStream) -> TokenStream {
17    let input = parse_macro_input!(input as TemplateInput);
18    let fn_name = input
19        .fn_name
20        .unwrap_or_else(|| default_fn_name(&input.path));
21    let suffix = format!("/rangular/{fn_name}.rs");
22    let suffix_lit = LitStr::new(&suffix, proc_macro2::Span::call_site());
23    quote! {
24        include!(concat!(env!("OUT_DIR"), #suffix_lit));
25    }
26    .into()
27}
28
29struct TemplateInput {
30    path: String,
31    fn_name: Option<String>,
32}
33
34impl syn::parse::Parse for TemplateInput {
35    fn parse(input: syn::parse::ParseStream<'_>) -> syn::Result<Self> {
36        let path_lit: LitStr = input.parse()?;
37        let path = path_lit.value();
38        let fn_name = if input.peek(syn::Token![,]) {
39            input.parse::<syn::Token![,]>()?;
40            let name_lit: LitStr = input.parse()?;
41            Some(name_lit.value())
42        } else {
43            None
44        };
45        Ok(Self { path, fn_name })
46    }
47}
48
49fn default_fn_name(path: &str) -> String {
50    PathBuf::from(path)
51        .file_stem()
52        .and_then(|stem| stem.to_str())
53        .unwrap_or("template")
54        .replace('-', "_")
55}

57#[cfg(test)]
58mod tests {
59    use super::{default_fn_name, TemplateInput};
60    use syn::parse_str;
61
62    #[test]
63    fn default_fn_name_replaces_hyphens() {
64        assert_eq!(default_fn_name("seed-bar.html"), "seed_bar");
65    }
66
67    #[test]
68    fn default_fn_name_fallbacks() {
69        assert_eq!(default_fn_name("plain.html"), "plain");
70        assert_eq!(default_fn_name(""), "template");
71        assert_eq!(default_fn_name("/tmp/foo-bar.ng.html"), "foo_bar.ng");
72    }
73
74    #[test]
75    fn template_input_parses_path_only_and_with_name() {
76        let path_only: TemplateInput = parse_str(r#""seed-bar.html""#).expect("path");
77        assert_eq!(path_only.path, "seed-bar.html");
78        assert!(path_only.fn_name.is_none());
79
80        let named: TemplateInput = parse_str(r#""seed-bar.html", "seed_bar_view""#).expect("named");
81        assert_eq!(named.path, "seed-bar.html");
82        assert_eq!(named.fn_name.as_deref(), Some("seed_bar_view"));
83    }
84}
