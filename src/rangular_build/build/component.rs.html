10impl Component {
11    pub(crate) fn scss_path(&self) -> Option<PathBuf> {
12        let path = self.directory.join(format!("{}.scss", self.name));
13        path.is_file().then_some(path)
14    }
15}

17pub fn discover(root: &Path) -> Result<Vec<Component>, String> {
18    let entries =
19        fs::read_dir(root).map_err(|error| format!("read {}: {error}", root.display()))?;
20    let mut components = Vec::new();
21    for entry in entries {
22        let entry = entry.map_err(|error| format!("read entry in {}: {error}", root.display()))?;
23        let directory = entry.path();
24        if !directory.is_dir() {
25            continue;
26        }
27
28        let name = file_name(&directory)?;
29        if directory.join(format!("{name}.html")).is_file() {
30            components.push(Component { name, directory });
31        }
32    }
33    components.sort_by(|left, right| left.name.cmp(&right.name));
34    if components.is_empty() {
35        return Err(format!("no components found under {}", root.display()));
36    }
37    Ok(components)
38}

40pub fn compile_template(
41    component: &Component,
42    manifest: &Path,
43) -> Result<(String, String), String> {
44    let html_path = component.directory.join(format!("{}.html", component.name));
45    let html = fs::read_to_string(&html_path)
46        .map_err(|error| format!("read {}: {error}", html_path.display()))?;
47    let source = relative_label(manifest, &html_path);
48    let view_name = format!("{}_view", component.name);
49    let compiled = rangular_aot::compile_named(&html, &source, &view_name);
50    if !compiled.ok() {
51        return Err(format!("{source}: {:?}", compiled.issues));
52    }
53    Ok((view_name, compiled.code))
54}

56pub fn relative_label(manifest: &Path, path: &Path) -> String {
57    path.strip_prefix(manifest)
58        .unwrap_or(path)
59        .to_string_lossy()
60        .into_owned()
61}
62
63fn file_name(path: &Path) -> Result<String, String> {
64    path.file_name()
65        .and_then(OsStr::to_str)
66        .map(str::to_owned)
67        .ok_or_else(|| format!("{} has no valid UTF-8 file name", path.display()))
68}

76    #[test]
77    fn discovers_sorted_template_directories_and_optional_scss() {
78        let temp = TempDir::new();
79        let root = temp.path().join("components");
80        fs::create_dir_all(root.join("zeta")).unwrap();
81        fs::write(root.join("zeta/zeta.html"), "<p>Zeta</p>").unwrap();
82        fs::write(root.join("zeta/zeta.scss"), ".zeta { color: red; }").unwrap();
83        fs::create_dir_all(root.join("alpha")).unwrap();
84        fs::write(root.join("alpha/alpha.html"), "<p>Alpha</p>").unwrap();
85        fs::create_dir_all(root.join("ignored")).unwrap();
86        fs::write(root.join("ignored/other.html"), "<p>Ignored</p>").unwrap();
87        fs::write(root.join("file.txt"), "ignored").unwrap();
88
89        let components = discover(&root).unwrap();
90        assert_eq!(
91            components
92                .iter()
93                .map(|c| c.name.as_str())
94                .collect::<Vec<_>>(),
95            ["alpha", "zeta"]
96        );
97        assert_eq!(components[0].scss_path(), None);
98        assert_eq!(components[1].scss_path(), Some(root.join("zeta/zeta.scss")));
99    }
