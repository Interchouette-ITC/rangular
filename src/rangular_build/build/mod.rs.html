15/// Discover and compile a package's rangular templates and styles.
16///
17/// Reads `CARGO_MANIFEST_DIR` and `OUT_DIR`, writes generated Rust under
18/// `OUT_DIR/rangular`, writes CSS to the configured output path, and emits Cargo
19/// rerun directives for the component directory and optional global stylesheet.
20/// Panics with the offending path and compiler diagnostics on failure, as
21/// expected for a Cargo build script.
22///
23/// # Arguments
24///
25/// * `config` - Component source directory, generated CSS path, and optional
26///   global SCSS path. Paths are relative to the Cargo package root.
27///
28/// # Panics
29///
30/// Panics when `CARGO_MANIFEST_DIR` or `OUT_DIR` is missing, when no matching
31/// components are found, when a template or stylesheet fails to compile, or
32/// when an input or output file cannot be read or written.
33///
34/// # Examples
35///
36/// In a package's `build.rs`:
37///
38/// ```no_run
39/// use rangular_build::{Config, build};
40///
41/// fn main() {
42///     build(Config::new("src/components", "style/components.css"));
43/// }
44/// ```
45#[allow(clippy::needless_pass_by_value)] // Keep the build-script API ergonomic for owned configs.
46pub fn build(config: Config) {
47    if let Err(error) = run(&config) {
48        panic!("rangular build failed: {error}");
49    }
50}

52fn run(config: &Config) -> Result<(), String> {
53    let manifest = env::var_os("CARGO_MANIFEST_DIR")
54        .map(PathBuf::from)
55        .ok_or_else(|| "CARGO_MANIFEST_DIR is not set".to_owned())?;
56    let out_dir = env::var_os("OUT_DIR")
57        .map(PathBuf::from)
58        .ok_or_else(|| "OUT_DIR is not set".to_owned())?;
59    run_at(config, &manifest, &out_dir)
60}

62fn run_at(
63    config: &Config,
64    manifest: &std::path::Path,
65    out_dir: &std::path::Path,
66) -> Result<(), String> {
67    let components_dir = manifest.join(&config.components_dir);
68    let generated_dir = out_dir.join("rangular");
69
70    let css_output = manifest.join(&config.css_output);
71
72    println!("cargo:rerun-if-changed={}", components_dir.display());
73    println!("cargo:rerun-if-changed={}", css_output.display());
74
75    if let Some(global_scss) = &config.global_scss {
76        println!(
77            "cargo:rerun-if-changed={}",
78            manifest.join(global_scss).display()
79        );
80    }
81
82    let components = discover(&components_dir)?;
83    fs::create_dir_all(&generated_dir)
84        .map_err(|error| format!("create {}: {error}", generated_dir.display()))?;
85
86    let mut css = String::from("/* Generated by rangular from component SCSS. */\n");
87    if let Some(global_scss) = &config.global_scss {
88        append_styles(
89            &manifest.join(global_scss),
90            &global_scss.display().to_string(),
91            &mut css,
92        )?;
93    }
94
95    let generated = compile_components(&components, manifest, &mut css)?;
96    for (view_name, code) in generated {
97        let output = generated_dir.join(format!("{view_name}.rs"));
98        fs::write(&output, code).map_err(|error| format!("write {}: {error}", output.display()))?;
99    }
100
101    if let Some(parent) = css_output.parent() {
102        fs::create_dir_all(parent)
103            .map_err(|error| format!("create {}: {error}", parent.display()))?;
104    }
105    fs::write(&css_output, css).map_err(|error| format!("write {}: {error}", css_output.display()))
106}

108fn compile_components(
109    components: &[Component],
110    manifest: &std::path::Path,
111    css: &mut String,
112) -> Result<Vec<(String, String)>, String> {
113    components
114        .iter()
115        .map(|component| {
116            let (view_name, code) = compile_template(component, manifest)?;
117            if let Some(scss_path) = component.scss_path() {
118                let label = relative_label(manifest, &scss_path);
119                let _ = writeln!(css, "\n/* {} */", component.name);
120                append_styles(&scss_path, &label, css)?;
121            }
122            Ok((view_name, code))
123        })
124        .collect()
125}
