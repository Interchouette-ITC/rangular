1use std::path::PathBuf;
2
3/// Build inputs and output locations for [`crate::build`].
4///
5/// Paths are relative to the Cargo package root. Component discovery checks
6/// immediate subdirectories for a matching `<directory-name>.html` template
7/// and optional `<directory-name>.scss` stylesheet.
8///
9/// # Examples
10///
11/// ```
12/// use rangular_build::Config;
13///
14/// let config = Config::new("src/components", "style/components.css")
15///     .global_scss("style/site.scss");
16/// ```
17#[derive(Clone, Debug)]
18pub struct Config {
19    pub(crate) components_dir: PathBuf,
20    pub(crate) css_output: PathBuf,
21    pub(crate) global_scss: Option<PathBuf>,
22}
23
24impl Config {
25    /// Create a configuration with a component directory and generated
26    /// stylesheet path.
27    ///
28    /// # Arguments
29    ///
30    /// * `components_dir` - Package-relative directory whose immediate child
31    ///   directories contain matching `<name>.html` templates and optional
32    ///   `<name>.scss` files. Commonly `src/components`.
33    /// * `css_output` - Package-relative destination for the combined CSS.
34    ///   Parent directories are created as needed; this path can point to a
35    ///   stylesheet loaded by Trunk.
36    ///
37    /// The returned configuration has no global stylesheet. Add one with
38    /// [`Self::global_scss`].
39    ///
40    /// # Examples
41    ///
42    /// ```
43    /// use rangular_build::Config;
44    ///
45    /// let config = Config::new("src/components", "style/components.css");
46    /// ```
47    #[must_use]
48    pub fn new(components_dir: impl Into<PathBuf>, css_output: impl Into<PathBuf>) -> Self {
49        Self {
50            components_dir: components_dir.into(),
51            css_output: css_output.into(),
52            global_scss: None,
53        }
54    }
55
56    /// Set a global SCSS file to prepend to the generated stylesheet.
57    ///
58    /// # Arguments
59    ///
60    /// * `path` - Package-relative SCSS file. It is compiled before stylesheets
61    ///   belonging to individual components.
62    ///
63    /// If the file is missing or contains invalid SCSS, [`crate::build`] reports
64    /// the path and compiler diagnostics and panics, as expected in a Cargo
65    /// build script.
66    ///
67    /// # Examples
68    ///
69    /// ```
70    /// use rangular_build::Config;
71    ///
72    /// let config = Config::new("src/components", "style/components.css")
73    ///     .global_scss("style/site.scss");
74    /// ```
75    #[must_use]
76    pub fn global_scss(mut self, path: impl Into<PathBuf>) -> Self {
77        self.global_scss = Some(path.into());
78        self
79    }
80}

82#[cfg(test)]
83mod tests {
84    use super::Config;
85    use std::path::PathBuf;
86
87    #[test]
88    fn stores_paths_and_optional_global_stylesheet() {
89        let plain = Config::new("components", "style/generated.css");
90        assert_eq!(plain.components_dir, PathBuf::from("components"));
91        assert_eq!(plain.css_output, PathBuf::from("style/generated.css"));
92        assert_eq!(plain.global_scss, None);
93
94        let configured = plain.global_scss("style/global.scss");
95        assert_eq!(
96            configured.global_scss,
97            Some(PathBuf::from("style/global.scss"))
98        );
99    }
