8/// Pure pipe transform: input value plus optional colon args.
9pub type PipeFn = fn(&Value, &[Value]) -> Result<Value, EvalError>;
10
11/// Stringly name → pipe function map (builtins + app custom).
12#[derive(Clone, Default)]
13pub struct PipeRegistry {
14    pipes: HashMap<String, PipeFn>,
15}
16
17impl PipeRegistry {
18    #[must_use]
19    pub fn new() -> Self {
20        Self::default()
21    }
22
23    /// Builtins: `uppercase`, `lowercase`, `number`, `json`.
24    #[must_use]
25    pub fn with_builtins() -> Self {
26        let mut reg = Self::new();
27        reg.register("uppercase", pipe_uppercase);
28        reg.register("lowercase", pipe_lowercase);
29        reg.register("number", pipe_number);
30        reg.register("json", pipe_json);
31        reg
32    }
33
34    /// Process-wide builtins registry.
35    #[must_use]
36    pub fn builtins() -> &'static Self {
37        static BUILTINS: OnceLock<PipeRegistry> = OnceLock::new();
38        BUILTINS.get_or_init(Self::with_builtins)
39    }
40
41    /// Shared builtins handle for `HostCell` / runtime.
42    #[must_use]
43    pub fn builtins_arc() -> Arc<Self> {
44        static BUILTINS_ARC: OnceLock<Arc<PipeRegistry>> = OnceLock::new();
45        BUILTINS_ARC
46            .get_or_init(|| Arc::new(Self::with_builtins()))
47            .clone()
48    }
49
50    pub fn register(&mut self, name: impl Into<String>, pipe: PipeFn) {
51        self.pipes.insert(name.into(), pipe);
52    }
53
54    /// # Errors
55    ///
56    /// When the pipe name is unknown or the transform rejects the value/args.
57    pub fn apply(&self, name: &str, value: &Value, args: &[Value]) -> Result<Value, EvalError> {
58        let pipe = self
59            .pipes
60            .get(name)
61            .ok_or_else(|| EvalError::UnknownPipe(name.to_owned()))?;
62        pipe(value, args)
63    }
64
65    #[must_use]
66    pub fn contains(&self, name: &str) -> bool {
67        self.pipes.contains_key(name)
68    }
69}

71fn display_plain(value: &Value) -> String {
72    match value {
73        Value::Str(s) => s.clone(),
74        Value::Num(n) => format_num(*n),
75        Value::Bool(b) => b.to_string(),
76        Value::List(items) => items
77            .iter()
78            .map(display_plain)
79            .collect::<Vec<_>>()
80            .join(","),
81        Value::Event(payload) => format!("{payload:?}"),
82        Value::Unit => String::new(),
83    }
84}

86fn format_num(n: f64) -> String {
87    if n.fract() == 0.0 && n.abs() < 1e15 {
88        format!("{n:.0}")
89    } else {
90        n.to_string()
91    }
92}
93
94fn as_number(value: &Value) -> Result<f64, EvalError> {
95    match value {
96        Value::Num(n) => Ok(*n),
97        Value::Str(s) => s
98            .parse::<f64>()
99            .map_err(|_| EvalError::TypeMismatch("number pipe")),
100        Value::Bool(b) => Ok(if *b { 1.0 } else { 0.0 }),
101        _ => Err(EvalError::TypeMismatch("number pipe")),
102    }
103}

105#[allow(clippy::unnecessary_wraps)] // PipeFn signature is always Result
106fn pipe_uppercase(value: &Value, _: &[Value]) -> Result<Value, EvalError> {
107    Ok(Value::Str(display_plain(value).to_uppercase()))
108}
109
110#[allow(clippy::unnecessary_wraps)] // PipeFn signature is always Result
111fn pipe_lowercase(value: &Value, _: &[Value]) -> Result<Value, EvalError> {
112    Ok(Value::Str(display_plain(value).to_lowercase()))
113}

115fn pipe_number(value: &Value, args: &[Value]) -> Result<Value, EvalError> {
116    let n = as_number(value)?;
117    if let Some(digits) = args.first() {
118        let raw = as_number(digits)?;
119        if !(0.0..=20.0).contains(&raw) || raw.fract() != 0.0 {
120            return Err(EvalError::TypeMismatch("number pipe digits"));
121        }
122        #[allow(
123            clippy::cast_possible_truncation,
124            clippy::cast_sign_loss,
125            reason = "digits validated to 0..=20 integer"
126        )]
127        let d = raw as usize;
128        return Ok(Value::Str(format!("{n:.d$}")));
129    }
130    Ok(Value::Str(format_num(n)))
131}

133#[allow(clippy::unnecessary_wraps)] // PipeFn signature is always Result
134fn pipe_json(value: &Value, _: &[Value]) -> Result<Value, EvalError> {
135    Ok(Value::Str(jsonish(value)))
136}

138fn jsonish(value: &Value) -> String {
139    match value {
140        Value::Str(s) => format!("\"{}\"", escape_json(s)),
141        Value::Num(n) => format_num(*n),
142        Value::Bool(b) => b.to_string(),
143        Value::List(items) => {
144            let inner = items.iter().map(jsonish).collect::<Vec<_>>().join(",");
145            format!("[{inner}]")
146        }
147        Value::Event(payload) => format!("\"{}\"", escape_json(&format!("{payload:?}"))),
148        Value::Unit => "null".into(),
149    }
150}
151
152fn escape_json(s: &str) -> String {
153    let mut out = String::with_capacity(s.len());
154    for ch in s.chars() {
155        match ch {
156            '"' => out.push_str("\\\""),
157            '\\' => out.push_str("\\\\"),
158            '\n' => out.push_str("\\n"),
159            '\r' => out.push_str("\\r"),
160            '\t' => out.push_str("\\t"),
161            c => out.push(c),
162        }
163    }
164    out
165}

171    #[test]
172    fn builtins_apply_case_and_json() {
173        let reg = PipeRegistry::with_builtins();
174        assert!(reg.contains("uppercase"));
175        assert_eq!(
176            reg.apply("uppercase", &Value::Str("ab".into()), &[])
177                .unwrap(),
178            Value::Str("AB".into())
179        );
180        assert_eq!(
181            reg.apply("lowercase", &Value::Str("AB".into()), &[])
182                .unwrap(),
183            Value::Str("ab".into())
184        );
185        assert_eq!(
186            reg.apply("json", &Value::Bool(true), &[]).unwrap(),
187            Value::Str("true".into())
188        );
189        assert!(matches!(
190            reg.apply("missing", &Value::Unit, &[]),
191            Err(EvalError::UnknownPipe(_))
192        ));
193    }

195    #[test]
196    fn number_pipe_formats_digits() {
197        let reg = PipeRegistry::with_builtins();
198        assert_eq!(

200                .unwrap(),
201            Value::Str("1.50".into())

203        assert!(matches!(
204            reg.apply("number", &Value::Str("x".into()), &[]),
