30fn scope_loop(scope: &Scope<'_>) -> TokenStream {
31    scope.loop_item.map_or_else(
32        || quote! { rangular_host::LoopScope::NONE },
33        |item| {
34            quote! {
35                rangular_host::LoopScope {
36                    item_name: Some(#item),
37                    item_val: Some(__rangular_loop.as_str()),
38                    index: Some(__rangular_index),
39                    count: Some(__rangular_count),
40                }
41            }
42        },
43    )
44}

46fn scoped_closure_body(scope: &Scope<'_>, invoke: &TokenStream) -> TokenStream {
47    scope.loop_item.map_or_else(
48        || quote! { move || #invoke },
49        |_item| {
50            quote! {
51                move || {
52                    let __rangular_loop = std::sync::Arc::clone(&__rangular_loop_store.get_value());
53                    let __rangular_index = __rangular_index;
54                    let __rangular_count = __rangular_count;
55                    #invoke
56                }
57            }
58        },
59    )
60}

62fn hoist_host_closure(
63    hoist: &mut HoistState,
64    scope: &Scope<'_>,
65    invoke: &TokenStream,
66) -> TokenStream {
67    let invoke = scoped_closure_body(scope, invoke);
68    hoist.hoist_closure(&invoke)

71fn hoist_event_closure(
72    hoist: &mut HoistState,
73    scope: &Scope<'_>,
74    event_name: &str,
75    ev_ty: &TokenStream,
76    body: &TokenStream,
77) -> TokenStream {
78    let ev = if matches!(event_name, "error" | "load") {
79        format_ident!("_ev")
80    } else {
81        format_ident!("ev")
82    };
83    let closure = scope.loop_item.map_or_else(
84        || {
85            quote! {
86                move |#ev: #ev_ty| {
87                    #body
88                }
89            }
90        },
91        |_item| {
92            quote! {
93                move |#ev: #ev_ty| {
94                    let __rangular_loop = std::sync::Arc::clone(&__rangular_loop_store.get_value());
95                    #body
96                }
97            }
98        },
99    );
100    hoist.hoist_closure(&closure)
101}

103struct HoistState {
104    expr_lets: Vec<TokenStream>,
105    closure_lets: Vec<TokenStream>,
106    head_mode: bool,
107}
108
109impl HoistState {
110    const fn new(head_mode: bool) -> Self {
111        Self {
112            expr_lets: Vec::new(),
113            closure_lets: Vec::new(),
114            head_mode,
115        }
116    }
117
118    fn prelude(&self) -> TokenStream {
119        let expr_lets = &self.expr_lets;
120        let closure_lets = &self.closure_lets;
121        quote! {
122            #(#expr_lets)*
123            #(#closure_lets)*
124        }
125    }
126
127    fn hoist_expr(&mut self, expr: &Expr) -> TokenStream {
128        let n = self.expr_lets.len();
129        let id = format_ident!("__rangular_e{n}");
130        let init = expr_tokens(expr);
131        self.expr_lets.push(quote! {
132            let #id: &'static rangular_expr::Expr = Box::leak(Box::new(#init));
133        });
134        quote! { #id }
135    }
136
137    fn hoist_closure(&mut self, closure: &TokenStream) -> TokenStream {
138        let n = self.closure_lets.len();
139        let id = format_ident!("__rangular_f{n}");
140        let closure = closure.clone();
141        self.closure_lets.push(quote! {
142            let #id = leptos::prelude::StoredValue::new(std::sync::Arc::new({
143                let host = __rangular_host.get_value();
144                #closure
145            }));
146        });
147        quote! { #id }
148    }
149
150    fn hoist_view_closure(&mut self, view: &TokenStream) -> TokenStream {
151        let n = self.closure_lets.len();
152        let id = format_ident!("__rangular_f{n}");
153        self.closure_lets.push(quote! {
154            let #id = leptos::prelude::StoredValue::new(std::sync::Arc::new(move || view! { #view }));
155        });
156        quote! { #id }
157    }
158}

178fn emit_rust_tokens_with_head_mode(
179    template: &Template,
180    fn_name: &str,
181    head_mode: bool,
182) -> EmitTokens {
183    let mut issues = Vec::new();
184    if template.nodes.is_empty() {
185        issues.push(AotIssue::error("RANG401", "empty template"));
186        return EmitTokens {
187            tokens: proc_macro2::TokenStream::new(),
188            issues,
189        };
190    }
191    let templates: HashMap<String, Vec<Node>> =
192        collect_ng_templates(&template.nodes).into_iter().collect();
193    let scope = Scope::root();
194    let mut hoist = HoistState::new(head_mode);
195    let Some(body) = lower_nodes(&template.nodes, &mut issues, &scope, &templates, &mut hoist)
196    else {
197        return EmitTokens {
198            tokens: proc_macro2::TokenStream::new(),
199            issues,
200        };
201    };
202    let prelude = hoist.prelude();
203    let ident = format_ident!("{fn_name}");
204    let selects = collect_projection_selects(&template.nodes);
205    let has_default = has_default_projection(&template.nodes);
206    let tokens = if selects.is_empty() && has_projection(&template.nodes) {
207        quote! {
208            pub fn #ident<H: rangular_host::Host + 'static>(
209                host: rangular_aot::HostCell<H>,
210                children: Children,
211            ) -> impl IntoView {
212                let __rangular_host = leptos::prelude::StoredValue::new(host);
213                #prelude
214                view! { #body }
215            }
216        }
217    } else if !selects.is_empty() {
218        let mut params = Vec::new();
219        for select in &selects {
220            let pname = format_ident!("{}", select_param_name(select));
221            params.push(quote! { #pname: Children });
222        }
223        if has_default {
224            params.push(quote! { children: Children });
225        }
226        quote! {
227            pub fn #ident<H: rangular_host::Host + 'static>(
228                host: rangular_aot::HostCell<H>,
229                #(#params),*
230            ) -> impl IntoView {
231                let __rangular_host = leptos::prelude::StoredValue::new(host);
232                #prelude
233                view! { #body }
234            }
235        }
236    } else {
237        quote! {
238            pub fn #ident<H: rangular_host::Host + 'static>(
239                host: rangular_aot::HostCell<H>,
240            ) -> impl IntoView {
241                let __rangular_host = leptos::prelude::StoredValue::new(host);
242                #prelude
243                view! { #body }
244            }
245        }
246    };
247    EmitTokens { tokens, issues }
248}

250fn has_projection(nodes: &[Node]) -> bool {
251    nodes.iter().any(|node| match node {
252        Node::Projection(_) => true,
253        Node::Element(el) => has_projection(&el.children),
254        Node::NgTemplate(t) => has_projection(&t.body),
255        Node::If(block) => {
256            has_projection(&block.then_branch)
257                || block
258                    .else_branch
259                    .as_ref()
260                    .is_some_and(|nodes| has_projection(nodes))
261        }
262        Node::For(block) => has_projection(&block.body),
263        Node::Text(_, _) | Node::Interpolation(_, _) | Node::Comment(_, _) => false,
264    })
265}

267fn lower_nodes(
268    nodes: &[Node],
269    issues: &mut Vec<AotIssue>,
270    scope: &Scope<'_>,
271    templates: &HashMap<String, Vec<Node>>,
272    hoist: &mut HoistState,
273) -> Option<TokenStream> {
274    if nodes.is_empty() {
275        return Some(quote! {});
276    }
277    let parts: Vec<_> = nodes
278        .iter()
279        .filter_map(|node| lower_node(node, issues, scope, templates, hoist))
280        .collect();
281    if parts.is_empty() {
282        if nodes
283            .iter()
284            .all(|n| matches!(n, Node::Comment(_, _) | Node::NgTemplate(_)))
285        {
286            return Some(quote! {});
287        }
288        issues.push(AotIssue::error("RANG401", "no lowerable template nodes"));
289        return None;
290    }
291    Some(quote! { #(#parts)* })
292}

294fn lower_node(
295    node: &Node,
296    issues: &mut Vec<AotIssue>,
297    scope: &Scope<'_>,
298    templates: &HashMap<String, Vec<Node>>,
299    hoist: &mut HoistState,
300) -> Option<TokenStream> {
301    if hoist.head_mode
302        && matches!(
303            node,
304            Node::Text(_, _) | Node::Interpolation(_, _) | Node::Projection(_)
305        )
306    {
307        issues.push(AotIssue::error(
308            "RANG403",
309            "text, interpolation, and projected content are not supported outside head elements",
310        ));
311        return None;
312    }
313
314    match node {
315        Node::Element(el) => lower_element(el, issues, scope, templates, hoist),
316        Node::Text(text, _) => {
317            let collapsed = text.split_whitespace().collect::<Vec<_>>().join(" ");
318            Some(quote! { #collapsed })
319        }
320        Node::Interpolation(expr, _) => {
321            let ex = hoist.hoist_expr(expr);
322            let loop_scope = scope_loop(scope);
323            let read = hoist_host_closure(
324                hoist,
325                scope,
326                &quote! { host.prop_str_scoped(#ex, #loop_scope) },
327            );
328            Some(quote! { {move || #read.get_value()()} })
329        }
330        Node::Comment(_, _) | Node::NgTemplate(_) => None,
331        Node::Projection(Projection { select, .. }) => select.as_ref().map_or_else(
332            || Some(quote! { {children()} }),
333            |sel| {
334                let pname = format_ident!("{}", select_param_name(sel));
335                Some(quote! { {#pname()} })
336            },
337        ),
338        Node::If(block) => lower_if(block, issues, scope, templates, hoist),
339        Node::For(block) => lower_for(block, issues, templates, hoist),
340    }
341}

343fn lower_element(
344    el: &Element,
345    issues: &mut Vec<AotIssue>,
346    scope: &Scope<'_>,
347    templates: &HashMap<String, Vec<Node>>,
348    hoist: &mut HoistState,
349) -> Option<TokenStream> {
350    let head_tag = hoist.head_mode.then(|| el.tag.to_ascii_lowercase());
351    if let Some(tag) = &head_tag {
352        if !matches!(
353            tag.as_str(),
354            "ng-container" | "title" | "meta" | "link" | "script"
355        ) {
356            issues.push(AotIssue::error(
357                "RANG403",
358                format!("unsupported element `<{}>` in head template", el.tag),
359            ));
360            return None;
361        }
362    }
363    if let Some(name) = template_outlet_ref(&el.attrs) {
364        let Some(body) = templates.get(name) else {
365            issues.push(AotIssue::error(
366                "RANG401",
367                format!("unknown ngTemplateOutlet ref `{name}`"),
368            ));
369            return None;
370        };
371        return lower_nodes(body, issues, scope, templates, hoist);
372    }
373    if el.tag == "ng-container" || head_tag.as_deref() == Some("ng-container") {
374        return lower_nodes(&el.children, issues, scope, templates, hoist);
375    }
376    if let Some(head_tag) = head_tag.as_deref() {
377        if head_tag == "title" {
378            return lower_head_title(el, issues, scope, hoist);
379        }
380        if head_tag == "meta" {
381            return lower_head_meta(el, issues, scope, hoist);
382        }
383        if head_tag == "link"
384            && static_attr_value(&el.attrs, "rel").is_some_and(|rel| {
385                rel.split_ascii_whitespace()
386                    .any(|token| token.eq_ignore_ascii_case("canonical"))
387            })
388        {
389            return lower_head_canonical_link(el, issues, scope, hoist);
390        }
391        if head_tag == "script" {
392            if has_static_attr_value(&el.attrs, "type", "application/ld+json") {
393                return lower_head_json_ld(el, issues, scope, hoist);
394            }
395            return Some(lower_head_script(el, scope, hoist));
396        }
397    }
398    let attrs = lower_attrs(&el.attrs, scope, hoist);
399    let children = lower_children(
400        &el.children,
401        el.self_closing,
402        issues,
403        scope,
404        templates,
405        hoist,
406    )?;
407    if el.tag.contains('-') {
408        let tag_lit = el.tag.as_str();
409        if el.self_closing {
410            Some(quote! {
411                <div data-rangular-component=#tag_lit #attrs />
412            })
413        } else {
414            Some(quote! {
415                <div data-rangular-component=#tag_lit #attrs>#children</div>
416            })
417        }
418    } else {
419        let tag = format_ident!("{}", sanitize_tag(&el.tag));
420        if el.self_closing {
421            Some(quote! { <#tag #attrs /> })
422        } else {
423            Some(quote! { <#tag #attrs>#children</#tag> })
424        }
425    }
426}

474fn lower_head_json_ld(
475    el: &Element,
476    issues: &mut Vec<AotIssue>,
477    scope: &Scope<'_>,
478    hoist: &mut HoistState,
479) -> Option<TokenStream> {
480    let type_attribute_count = el
481        .attrs
482        .iter()
483        .filter(|attr| match attr {
484            Attr::Static { name, .. }
485            | Attr::Property { name, .. }
486            | Attr::Attribute { name, .. }
487            | Attr::Input { name, .. } => name.eq_ignore_ascii_case("type"),
488            Attr::Class { .. } | Attr::Event { .. } | Attr::Output { .. } | Attr::Ref { .. } => {
489                false
490            }
491        })
492        .count();
493    if type_attribute_count != 1
494        || has_dynamic_attr(&el.attrs, "type")
495        || !static_attr_value(&el.attrs, "type")
496            .is_some_and(|value| value.eq_ignore_ascii_case("application/ld+json"))
497    {
498        issues.push(AotIssue::error(
499            "RANG403",
500            "JSON-LD script elements support only the static type attribute",
501        ));
502        return None;
503    }
504    let dynamic_values = el
505        .children
506        .iter()
507        .filter(|node| matches!(node, Node::Interpolation(_, _)))
508        .count();
509    let has_non_whitespace_text = el
510        .children
511        .iter()
512        .any(|node| matches!(node, Node::Text(text, _) if !text.trim().is_empty()));
513    if dynamic_values > 1 || (dynamic_values == 1 && has_non_whitespace_text) {
514        issues.push(AotIssue::error(
515            "RANG403",
516            "dynamic JSON-LD must be supplied as one complete JSON interpolation",
517        ));
518        return None;
519    }
520    let json = lower_head_text_content(&el.children, issues, scope, hoist, false)?;
521    let attrs = lower_attrs(&el.attrs, scope, hoist);
522    Some(quote! {
523        <script #attrs>
524            {move || {
525                let __rangular_json = (#json)();
526                rangular_aot::escape_json_ld(&__rangular_json)
527            }}
528        </script>
529    })
530}

532fn lower_head_script(el: &Element, scope: &Scope<'_>, hoist: &mut HoistState) -> TokenStream {
533    let raw_text = el
534        .children
535        .iter()
536        .filter_map(raw_script_text)
537        .collect::<String>();

550fn lower_head_text_content(
551    nodes: &[Node],
552    issues: &mut Vec<AotIssue>,
553    scope: &Scope<'_>,
554    hoist: &mut HoistState,
555    normalize_whitespace: bool,
556) -> Option<TokenStream> {
557    let mut format = String::new();
558    let mut values = Vec::new();
559    let has_expression = nodes
560        .iter()
561        .any(|node| matches!(node, Node::Interpolation(_, _)));
562    for node in nodes {
563        match node {
564            Node::Text(text, _) => {
565                let text = if normalize_whitespace && !has_expression {
566                    text.split_whitespace().collect::<Vec<_>>().join(" ")
567                } else {
568                    text.clone()
569                };
570                if has_expression {
571                    format.push_str(&text.replace('{', "{{").replace('}', "}}"));
572                } else {
573                    format.push_str(&text);
574                }
575            }
576            Node::Interpolation(expr, _) => {
577                format.push_str("{}");
578                let value = lower_head_binding(expr, scope, hoist);
579                values.push(quote! { (#value)() });
580            }
581            Node::Comment(_, _) => {}
582            _ => {
583                issues.push(AotIssue::error(
584                    "RANG403",
585                    "head title and JSON-LD content support only text and interpolations",
586                ));
587                return None;
588            }
589        }
590    }
591    if !has_expression {
592        return Some(quote! { move || String::from(#format) });
593    }
594    Some(quote! { move || format!(#format, #(#values),*) })
595}

597fn lower_head_binding(expr: &Expr, scope: &Scope<'_>, hoist: &mut HoistState) -> TokenStream {
598    let expr = hoist.hoist_expr(expr);
599    let loop_scope = scope_loop(scope);
600    let value = hoist_host_closure(
601        hoist,
602        scope,
603        &quote! { host.prop_str_scoped(#expr, #loop_scope) },
604    );
605    quote! { move || #value.get_value()() }
606}

608fn static_attr_value<'a>(attrs: &'a [Attr], name: &str) -> Option<&'a str> {
609    attrs.iter().find_map(|attr| match attr {
610        Attr::Static {
611            name: attr_name,
612            value: Some(value),
613            ..
614        } if attr_name.eq_ignore_ascii_case(name) => Some(value.as_str()),
615        _ => None,
616    })

619fn has_static_attr_value(attrs: &[Attr], name: &str, value: &str) -> bool {

622            attr,
623            Attr::Static {
624                name: attr_name,
625                value: Some(attr_value),
626                ..
627            } if attr_name.eq_ignore_ascii_case(name)
628                && attr_value.eq_ignore_ascii_case(value)

630    })

657fn lower_attrs(attrs: &[Attr], scope: &Scope<'_>, hoist: &mut HoistState) -> TokenStream {
658    let mut tokens = Vec::new();
659    for attr in attrs {
660        if matches!(attr, Attr::Ref { .. }) {
661            continue;
662        }
663        if matches!(
664            attr,
665            Attr::Property {
666                name,
667                ..
668            } if name == "ngTemplateOutlet"
669        ) {
670            continue;
671        }
672        tokens.push(lower_one_attr(attr, scope, hoist));
673    }
674    if tokens.is_empty() {
675        quote! {}
676    } else {
677        quote! { #(#tokens)* }
678    }
679}

681fn lower_one_attr(attr: &Attr, scope: &Scope<'_>, hoist: &mut HoistState) -> TokenStream {
682    let loop_scope = scope_loop(scope);
683    match attr {
684        Attr::Static {
685            name,
686            value: Some(value),
687            ..
688        } => static_attr(name, value),
689        Attr::Static {
690            name, value: None, ..
691        } => html_name(name),
692        Attr::Property { name, expr, .. } if name == "disabled" || name == "checked" => {
693            let ex = hoist.hoist_expr(expr);
694            let handler = hoist_host_closure(
695                hoist,
696                scope,
697                &quote! { host.eval_bool_scoped(#ex, #loop_scope) },
698            );
699            prop_attr(name, &handler)
700        }
701        Attr::Property { name, expr, .. } => {
702            let ex = hoist.hoist_expr(expr);
703            let handler = hoist_host_closure(
704                hoist,
705                scope,
706                &quote! { host.prop_str_scoped(#ex, #loop_scope) },
707            );
708            prop_attr(name, &handler)
709        }
710        Attr::Attribute { name, expr, .. } | Attr::Input { name, expr, .. } => {
711            let attr_name = match attr {
712                Attr::Input { name, .. } => format!("data-input-{name}"),
713                _ => name.clone(),
714            };
715            let ex = hoist.hoist_expr(expr);
716            let handler = hoist_host_closure(
717                hoist,
718                scope,
719                &quote! { host.prop_str_scoped(#ex, #loop_scope) },
720            );
721            attr_binding(&attr_name, &handler)
722        }
723        Attr::Class { name, expr, .. } => {
724            let ex = hoist.hoist_expr(expr);
725            let handler = hoist_host_closure(
726                hoist,
727                scope,
728                &quote! { host.eval_truthy_scoped(#ex, #loop_scope) },
729            );
730            class_attr(name, &handler)
731        }
732        Attr::Event { name, expr, .. } => lower_dom_event(name, expr, scope, hoist),
733        Attr::Output { name, expr, .. } => {
734            let attr_name = format!("data-output-{name}");
735            let handler = rangular_parser::event_handler_name(expr);
736            static_attr(&attr_name, handler)
737        }
738        Attr::Ref { .. } => quote! {},
739    }
740}

742fn event_value_tokens(event_name: &str) -> TokenStream {
743    match event_name {
744        "error" | "load" => quote! { String::new() },
745        "click" | "dblclick" | "auxclick" => {
746            quote! { format!("{},{}", ev.client_x(), ev.client_y()) }
747        }
748        "change" => quote! {
749            {
750                use wasm_bindgen::JsCast;
751                ev.target()
752                    .and_then(|t| t.dyn_into::<web_sys::HtmlInputElement>().ok())
753                    .map(|el| {
754                        let ty = el.type_();
755                        if ty == "checkbox" || ty == "radio" {
756                            el.checked().to_string()
757                        } else {
758                            el.value()
759                        }
760                    })
761                    .or_else(|| {
762                        ev.target()
763                            .and_then(|t| t.dyn_into::<web_sys::HtmlSelectElement>().ok())
764                            .map(|el| el.value())
765                    })
766                    .unwrap_or_default()
767            }
768        },
769        _ => quote! {
770            {
771                use wasm_bindgen::JsCast;
772                ev.target()
773                    .and_then(|t| t.dyn_into::<web_sys::HtmlInputElement>().ok())
774                    .map(|el| el.value())
775                    .unwrap_or_default()
776            }
777        },
778    }
779}

789fn lower_dom_event(
790    name: &str,
791    expr: &Expr,
792    scope: &Scope<'_>,
793    hoist: &mut HoistState,
794) -> TokenStream {
795    let handler = rangular_parser::event_handler_name(expr);
796    let ex = hoist.hoist_expr(expr);
797    let loop_scope = scope_loop(scope);
798    let ev_ty = event_param_type(name);
799    let event_value = event_value_tokens(name);
800    let callback = hoist_event_closure(
801        hoist,
802        scope,
803        name,
804        &ev_ty,
805        &quote! {
806            let event_value = #event_value;
807            host.emit_dom_event_call_scoped(#handler, #ex, #name, event_value, #loop_scope);
808        },
809    );
810    event_attr(name, &callback)
811}

813fn lower_if(
814    block: &IfBlock,
815    issues: &mut Vec<AotIssue>,
816    scope: &Scope<'_>,
817    templates: &HashMap<String, Vec<Node>>,
818    hoist: &mut HoistState,
819) -> Option<TokenStream> {
820    let cond = hoist.hoist_expr(&block.cond);
821    let loop_scope = scope_loop(scope);
822    let then_view = lower_nodes(&block.then_branch, issues, scope, templates, hoist)?;
823    if let Some(else_branch) = &block.else_branch {
824        let else_view = lower_nodes(else_branch, issues, scope, templates, hoist)?;
825        let when = hoist_host_closure(
826            hoist,
827            scope,
828            &quote! { host.eval_truthy_scoped(#cond, #loop_scope) },
829        );
830        let fallback = hoist.hoist_view_closure(&quote! { #else_view });
831        Some(quote! {
832            <Show
833                when=move || #when.get_value()()
834                fallback=move || #fallback.get_value()()
835            >
836                #then_view
837            </Show>
838        })
839    } else {
840        let when = hoist_host_closure(
841            hoist,
842            scope,
843            &quote! { host.eval_truthy_scoped(#cond, #loop_scope) },
844        );
845        Some(quote! {
846            <Show when=move || #when.get_value()()>
847                #then_view
848            </Show>
849        })
850    }
851}

853fn lower_for(
854    block: &ForBlock,
855    issues: &mut Vec<AotIssue>,
856    templates: &HashMap<String, Vec<Node>>,
857    hoist: &mut HoistState,
858) -> Option<TokenStream> {
859    let iter = hoist.hoist_expr(&block.iter);
860    let item_name = block.item.as_str();
861    let item_ident = format_ident!("{}", item_name);
862    let scope = Scope::with_loop_item(item_name);
863    let mut body_hoist = HoistState::new(hoist.head_mode);
864    let body = lower_nodes(&block.body, issues, &scope, templates, &mut body_hoist)?;
865    let body_prelude = body_hoist.prelude();
866    let each = hoist.hoist_closure(&quote! {
867        move || {
868            let list = host.eval_list(#iter);
869            let count = list.len();
870            list.into_iter()
871                .enumerate()
872                .map(|(index, item)| (index, count, item))
873                .collect::<Vec<(usize, usize, String)>>()
874        }
875    });
876    let item_lit = item_name;
877    let key = block.track.as_ref().map_or_else(
878        || {
879            quote! {
880                |row: &(usize, usize, String)| {
881                    let (_, _, item) = row;
882                    item.clone()
883                }
884            }
885        },
886        |track| {
887            let tr = hoist.hoist_expr(track);
888            let key_fn = hoist.hoist_closure(&quote! {
889                move |row: &(usize, usize, String)| {
890                    let (__rangular_index, __rangular_count, item) = row;
891                    host.prop_str_scoped(
892                        #tr,
893                        rangular_host::LoopScope {
894                            item_name: Some(#item_lit),
895                            item_val: Some(item.as_str()),
896                            index: Some(*__rangular_index),
897                            count: Some(*__rangular_count),
898                        },
899                    )
900                }
901            });
902            quote! {
903                move |row: &(usize, usize, String)| {
904                    #key_fn.get_value()(row)
905                }
906            }
907        },
908    );
909    Some(quote! {
910        <For
911            each=move || #each.get_value()()
912            key=#key
913            let:row
914        >
915            {
916                let (__rangular_index, __rangular_count, #item_ident) = row;
917                let __rangular_loop_store =
918                    leptos::prelude::StoredValue::new(std::sync::Arc::new(#item_ident));
919                #body_prelude
920                view! { #body }
921            }
922        </For>
923    })
924}

926fn html_name(name: &str) -> TokenStream {
927    let mut out = TokenStream::new();
928    let mut first = true;
929    let mut after_empty = false;
930    for part in name.split('-') {
931        if part.is_empty() {
932            out.extend([TokenTree::Punct(Punct::new('-', Spacing::Joint))]);
933            out.extend([TokenTree::Punct(Punct::new('-', Spacing::Joint))]);
934            after_empty = true;
935            continue;
936        }
937        if !first && !after_empty {
938            out.extend([TokenTree::Punct(Punct::new('-', Spacing::Joint))]);
939        }
940        after_empty = false;
941        out.extend([TokenTree::Ident(format_ident!("{part}"))]);
942        first = false;
943    }
944    out
945}

947fn static_attr(name: &str, value: &str) -> TokenStream {
948    if name == "property" {
949        return custom_attr(name, &quote! { #value });
950    }
951    let mut out = html_name(name);
952    let eq = Punct::new('=', Spacing::Joint);
953    out.extend([TokenTree::Punct(eq)]);
954    out.extend([TokenTree::Literal(Literal::string(value))]);
955    out
956}
957
958fn prefix_attr(prefix: &str, name: &str, value: &TokenStream) -> TokenStream {
959    let mut out = TokenStream::new();
960    out.extend([TokenTree::Ident(format_ident!("{prefix}"))]);
961    let colon = Punct::new(':', Spacing::Joint);
962    out.extend([TokenTree::Punct(colon)]);
963    out.extend(html_name(name));
964    let eq = Punct::new('=', Spacing::Joint);
965    out.extend([TokenTree::Punct(eq)]);
966    out.extend(value.clone());
967    out
968}
969
970fn prop_attr(name: &str, value: &TokenStream) -> TokenStream {
971    let handler = quote! { move || #value.get_value()() };
972    prefix_attr("prop", name, &handler)
973}
974
975fn attr_binding(name: &str, value: &TokenStream) -> TokenStream {
976    let handler = quote! { move || #value.get_value()() };
977    if name == "property" {
978        return custom_attr(name, &handler);
979    }
980    let mut out = html_name(name);
981    out.extend([TokenTree::Punct(Punct::new('=', Spacing::Joint))]);
982    out.extend(handler);
983    out
984}

986fn custom_attr(name: &str, value: &TokenStream) -> TokenStream {
987    quote! {
988        {..leptos::tachys::html::attribute::custom::custom_attribute(#name, #value)}
989    }
990}
991
992fn class_attr(name: &str, value: &TokenStream) -> TokenStream {
993    let lit = Literal::string(name);
994    let handler = quote! { move || #value.get_value()() };
995    quote! { class=(#lit, #handler) }
996}

998fn event_attr(name: &str, value: &TokenStream) -> TokenStream {
999    let handler = quote! { move |ev| #value.get_value()(ev) };
1000    prefix_attr("on", name, &handler)
1001}
